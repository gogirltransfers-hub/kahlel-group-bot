def ask_ai(user_message, business_code):
    brain = next((b["brain"] for b in BUSINESSES.values() if b["code"]==business_code), "You are helpful.")
    try:
        completion = client.chat.completions.create(
            model="llama-3.3-70b-versatile",
            messages=[{"role": "system", "content": brain}, {"role": "user", "content": user_message}],
            temperature=0.7, max_tokens=350
        )
        return completion.choices[0].message.content
    except:

def send_whatsapp_d7(to_number, message_text):
    url = "https://api.d7networks.com/whatsapp/v2/send"
    headers = {"Authorization": f"Bearer {D7_API_TOKEN}", "Content-Type": "application/json"}
    payload = {"to": to_number, "from": D7_WHATSAPP_NUMBER, "type": "text", "content": {"text": message_text}}
    try:
        r = requests.post(url, json=payload, headers=headers, timeout=15)
        if r.status_code == 404:
            url2 = "https://api.d7networks.com/whatsapp/v1/messages"
            payload2 = {"originator": D7_WHATSAPP_NUMBER, "recipient": to_number, "content": message_text, "message_type": "text"}
            r = requests.post(url2, json=payload2, headers=headers, timeout=15)
        return r.ok
    except: return False

def detect_intent(text):
    text_lower = text.lower().strip()
    if text_lower in BUSINESSES: return BUSINESSES[text_lower]["code"]
    if text_lower and text_lower[0] in BUSINESSES and len(text_lower)<=2: return BUSINESSES[text_lower[0]]["code"]
    for biz in BUSINESSES.values():
        for kw in biz["keywords"]:
            if kw in text_lower: return biz["code"]
    return None

def get_admin_stats():
    today = datetime.now().strftime("%Y-%m-%d")
    total = stats["total_chats"]
    by_biz = stats["by_business"]
    recent = stats["recent_leads"][-10:]
    msg = f"📊 *KAHLEL GROUP ADMIN DASHBOARD*\nDate: {today}\n\n"
    msg += f"Total Chats Today: {stats['by_day'][today]}\nTotal All Time: {total}\n\n"
    msg += "*By Business:*\n"
    for code, count in by_biz.most_common():
        biz_name = next((b["name"] for b in BUSINESSES.values() if b["code"]==code), code)
        msg += f"{biz_name}: {count}\n"
    if not by_biz:
        msg += "No chats yet today\n"
    msg += f"\n*Recent Leads (last 10):*\n"
    for lead in reversed(recent):
        msg += f"{lead['time']} - {lead['number']} -> {lead['business']}: {lead['msg'][:30]}...\n"
    msg += f"\nActive Sessions: {len(user_sessions)}\n\nType ADMIN RESET to clear stats"
    return msg

@app.route("/webhook", methods=["GET", "POST"])
def webhook():
    if request.method == "GET":
        return request.args.get("hub.challenge", "Kahlel Group Active"), 200
    try:
        data = request.get_json(force=True)
        from_number = None
        user_text = None
        if isinstance(data, dict):
            from_number = data.get("sender") or data.get("from") or data.get("originator") or data.get("waId")
            user_text = data.get("text") or data.get("body") or data.get("content") or data.get("message")
            if isinstance(user_text, dict):
                user_text = user_text.get("text") or user_text.get("body")
            if not user_text and "message" in data:
                user_text = data["message"].get("text") if isinstance(data["message"], dict) else data["message"]
        if not from_number or not user_text: return jsonify({"status":"ignored"}),200
        from_number = str(from_number).replace(" ","").replace("-","")
        if not from_number.startswith("+"):
            if from_number.startswith("264"): from_number = "+"+from_number
        user_text = str(user_text).strip()
        user_text_lower = user_text.lower()
        now = datetime.now()

        # === ADMIN COMMANDS ===
        if user_text_lower.startswith("admin"):
            # Check if admin (simple check - you can add your personal number to ADMIN_NUMBERS)
            is_admin = True  # For now allow anyone with PIN, change to check number later
            if ADMIN_PIN in user_text or is_admin:
                if "stats" in user_text_lower or user_text_lower=="admin":
                    send_whatsapp_d7(from_number, get_admin_stats())
                    return jsonify({"status":"admin stats sent"}),200
                elif "reset" in user_text_lower:
                    stats["total_chats"]=0
                    stats["by_business"].clear()
                    stats["by_day"].clear()
                    stats["recent_leads"].clear()
                    send_whatsapp_d7(from_number, "✅ Stats reset! Dashboard cleared.")
                    return jsonify({"status":"reset"}),200
                elif "leads" in user_text_lower:
                    leads = stats["recent_leads"][-20:]
                    msg = "📋 *Last 20 Leads:*\n" + "\n".join([f"{l['number']} -> {l['business']}" for l in leads]) or "No leads"
                    send_whatsapp_d7(from_number, msg)
                    return jsonify({"status":"leads sent"}),200
            else:
                send_whatsapp_d7(from_number, "🔒 Admin access denied. Need PIN.")
                return jsonify({"status":"admin denied"}),200

        if from_number in user_sessions and now - user_sessions[from_number]['time'] > timedelta(hours=24):
            del user_sessions[from_number]

        if user_text_lower in ["menu","start","hi","hello","back","change","0","group"]:
            if from_number in user_sessions: del user_sessions[from_number]
            send_whatsapp_d7(from_number, get_welcome_message())
            return jsonify({"status":"menu sent"}),200

        if from_number not in user_sessions:
            intent_code = detect_intent(user_text)
            if intent_code:
                user_sessions[from_number] = {'intent': intent_code, 'time': now}
                stats["total_chats"]+=1
                stats["by_business"][intent_code]+=1
                stats["by_day"][now.strftime("%Y-%m-%d")]+=1
                stats["recent_leads"].append({"number":from_number,"business":intent_code,"msg":user_text,"time":now.strftime("%H:%M")})
                if len(stats["recent_leads"])>50: stats["recent_leads"].pop(0)
                biz_name = next((b['name'] for b in BUSINESSES.values() if b['code']==intent_code), "Kahlel")
                icon = next((b['icon'] for b in BUSINESSES.values() if b['code']==intent_code), "")
                reply = f"Great! Switched to *{biz_name}* {icon}\n\n" + ask_ai(user_text, intent_code)
                send_whatsapp_d7(from_number, reply)
                return jsonify({"status":f"routed to {intent_code}"}),200
            else:
                send_whatsapp_d7(from_number, get_welcome_message())
                return jsonify({"status":"welcome"}),200
        else:
            intent_code = user_sessions[from_number]['intent']
            user_sessions[from_number]['time']=now
            new_intent = detect_intent(user_text)
            if new_intent and new_intent!=intent_code and len(user_text)<15:
                user_sessions[from_number]['intent']=new_intent
                intent_code=new_intent
                stats["by_business"][intent_code]+=1
                biz_name = next((b['name'] for b in BUSINESSES.values() if b['code']==intent_code), "Kahlel")
                reply = f"Switched to *{biz_name}*! 🔄\n\n" + ask_ai(user_text, intent_code)
                send_whatsapp_d7(from_number, reply)
                return jsonify({"status":f"switched to {intent_code}"}),200
            ai_reply = ask_ai(user_text, intent_code)
            send_whatsapp_d7(from_number, ai_reply)
            return jsonify({"status":"sent"}),200
    except Exception as e:
        print(f"Error: {e}")
        import traceback; traceback.print_exc()
        return jsonify({"error": str(e)}),200
